\section{Connectionism}

\begin{definition}[MP Neuron]
    \(\x \in \{0, 1\}^n, \sig \in \{-1, +1\}^n, \theta \in \R\):
    \(f[\sig, \theta](\x) = \vb{1}\{\sig \cdot \x \geq \theta\}\)
\end{definition}

\begin{definition}[Thresh. Unit]
    \(\x \in \R^n, \w \in \R^n, b \in \R\):
    \(f[\w, b](\x) = \sign(\x \cdot \w + b)\)
\end{definition}

\begin{definition}[Hesse]
    \(\x \cdot \w + b \overset{!}{=} 0 \iff \frac{\x \cdot \w}{\norm{\w}} + \frac{b}{\norm{\w}} = 0\), \(\n = \frac{\w}{\norm{\w}}\). \(- b / \norm{\w}\) dist. to origin.
\end{definition}

\begin{definition}[Geometric Margin]
    sign dist: \({\color{H2}\gamma[\w, b](\x, y)} := \frac{y(\x \cdot \w + b)}{\norm{\w}}\). Correct if: \[f[\w, b](\x) = y \iff \sign(\x \cdot \w + b) = y \iff \sign({\color{H2}\gamma}) = 1 \iff {\color{H2}\gamma} > 0\].
\end{definition}

\vspace{-1em}
\begin{definition}[Classifier Margin]
    \({\color{H3}\gamma[\w, b](S)} := \min_{(\x, y) \in S} {\color{H2}\gamma[\w, b](\x, y)}\)
\end{definition}

\begin{definition}[Max Margin]
    \(\argmax{\w, b} {\color{H3}\gamma[\w, b](S)}\). \adv{unique up to positive scaling}.
\end{definition}

\begin{definition}[Version Space]
    \(\VS(S) = \{\theta \mid \forall (\x, y) \in S: f[\theta](\x) = y\} \neq \emptyset \implies\) Sep. \(S\)
\end{definition}

\begin{definition}[Perc. Learn.]
    \resizebox{.80\linewidth}{!}{
    \(f[\w, b](\x) \neq y \implies \w \mathrel{+}= y\x, b \mathrel{+}= y\). \(\w^0 \in \text{span}\{\x_i\} \implies \w^t \in \text{span}\{\x_i\}\).
    }
\end{definition}

\begin{lemma}
    \({\color{H3}\gamma[\w](S)} = {\color{H3}\gamma} > 0, \norm{\w} = 1 \implies \w^t \cdot \w \geq t {\color{H3}\gamma}\).
\end{lemma}

\begin{lemma}
    \(R = \max_{\x \in S} \norm{\x} \implies \norm{\w^t} \leq R\sqrt{t}\).
\end{lemma}

\begin{theorem}[Navikoff]
    \(S\) is \({\color{H3}\gamma}\)-sep \(\implies\) converges in at most \(\lfloor R^2 / {\color{H3}\gamma}^2\rfloor\) steps.
\end{theorem}

    \(C(s+1, n) = 2 \sum_{i=0}^{n-1}\binom{s}{i}\). Rec:

    }
\end{theorem}

\begin{definition}[Delta Learning]
    Function \(f[W, b](x) = \phi(Wx + b)\).
    \(\Delta w_{ij} = \eta \delta_i \phi'(w_i \cdot \x_j)\), \(\delta_i := (y_i - \phi_i)\), \(\eta > 0\). Derived: \(l = \frac{1}{2} \sum_{i=1}^n (\phi_i - y_i)^2, \phi_i := \phi(w_i \cdot \x)\) \(\equiv\) SGD.
\end{definition}

\subsection{Hopfield Networks}

\textbf{Energy Function}: \(E(X) = - \frac{1}{2} \sum_{i \neq j} w_{ij} X_i X_j + \sum_i b_i X_i, X_i \in \{\pm 1\}.\) with constraint \(w_{ij} = w_{ji}\) and \(w_{ii}\) = 0.
Dynamics: \(X_i \gets 1\) if \(E(X_i = 1) \leq E(X_i = -1)\).

Can be be updated \textit{synchronously} or \textit{asynchronously}. Energy always decreases or stays the same. Converges to local minimum.

\begin{definition}[Effective Field]
    \(H_i := \sum_j w_{ij} X_j - b_i\). Update: \(X_i \gets \sign(H_i) \setminus \{0\}\).
\end{definition}

\begin{definition}[Hebb Learning]
    Patterns $x^t \in \{\pm 1\}^n, 1 \leq t \leq s \Rightarrow W = \frac{1}{n} \sum_{t=1}^s x^t {x^t}^\top$.
    Meta-stable: $x_i^t \overset{!}{=} \sign\big(\sum_{j=1}^n w_{ij} x_j^t\big)$. 
    $\sum w_{ij} x_j^t = \underbrace{x_i^t}_{\text{Signal}} + \underbrace{\frac{1}{n} \sum_{j, r \neq t} x_i^r x_j^r x_j^t}_{:= C_i^t (\text{Cross-talk})}$.

    \vspace{-1.5em}
    \textit{If $|C_i^t| > 1$, noise flips the bit.}
\end{definition}

\begin{definition}[Capacity]
    Assume iid random signs, $n \to \infty$. By CLT, $C_i^t \sim \mathcal{N}(0, \frac{s}{n})$.
    \textit{Intuition: Sum of $n(s-1)$ independent $\pm \frac{1}{n}$ terms $\Rightarrow$ Var $\approx \frac{ns}{n^2} = \frac{s}{n}$.}

    Requiering a pattern to be retrieved with high porobability: \(s \leq \frac{n}{2 \log_2 n}\)
\end{definition}
